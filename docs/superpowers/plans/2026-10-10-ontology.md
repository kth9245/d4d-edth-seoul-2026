# 작전 온톨로지 + 객체 그래프 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 기체 능력을 is-a 종류 계층에서 추론해 플레이 검사와 편집기에 경로로 보이고, 실행 기록에서 객체 그래프를 만들어 상황판 카드에서 연다.

**Architecture:** 계층 데이터는 시연 시나리오 JSON의 `ontology` 칸. 새 모듈 `modules/ontology`가 추론(체인·판정·채우기·화면용 판정표)을 하고, sim 로더가 시나리오를 읽을 때 기체 `capabilities`·`kind_path`를 채운다. gateway `EngineState`에 결정·명령 기록을 쌓아 `build_object_graph`가 `GET /graph`로, `ontology_view`가 `GET /ontology`로 내보낸다. 화면은 편집기 사이드바 "온톨로지"와 상황판 카드 "연결 보기"(React Flow).

**Tech Stack:** Python 3 · pydantic v2 · FastAPI · pytest(xdist) · React 19 · @xyflow/react 12 · vitest

**Spec:** `docs/superpowers/specs/2026-10-10-ontology-design.md` (결정 0038, 할 일 Part1-24)

```python
"""Ontology (decision 0038): vehicle kinds in an is-a hierarchy whose capabilities subkinds inherit,
the reasoning shown to the operator, and the graph of objects linked in a run."""

from typing import Literal

from pydantic import BaseModel

from .enums import ActionKind


class ActionGroup(BaseModel):  # e.g. watch: watch_target, watch_point, search_area
    id: str
    label: str
    actions: list[ActionKind]


class VehicleKind(BaseModel):
    id: str
    label: str
    is_a: str | None = None  # parent kind id; None = the root
    provides: list[str] = []  # ActionGroup ids or ActionKind values this kind adds


class Ontology(BaseModel):
    groups: list[ActionGroup]
    kinds: list[VehicleKind]


class CapabilityReason(BaseModel):
    action: ActionKind
    allowed: bool
    path: list[str]  # kind ids from the vehicle's kind up to the giver (allowed) or the root (denied)
    via: str | None = None  # the provides entry that matched


class VehicleCapabilities(BaseModel):
    vehicle_id: str
    kind: str
    reasons: list[CapabilityReason]  # one per ActionKind, in enum order


class OntologyView(BaseModel):  # GET /ontology
    ontology: Ontology
    vehicles: list[VehicleCapabilities]


class ObjectNode(BaseModel):
    id: str
    type: Literal["vehicle", "target", "report", "decision", "command", "block"]
    label: str
    t_ms: int | None = None


class ObjectLink(BaseModel):
    src: str
    dst: str
    rel: Literal["reported", "about", "answers", "from_decision", "sent_to", "aims_at", "blocked"]


class ObjectGraph(BaseModel):  # GET /graph?focus=
    focus: str
    nodes: list[ObjectNode]
    links: list[ObjectLink]
```

```python
from contracts.models import ActionKind, CapabilityReason, Ontology
from modules.ontology.actions.kind_chain import kind_chain


def explain_capability(kind: str, action: ActionKind, onto: Ontology) -> CapabilityReason:
    groups = {g.id: g.actions for g in onto.groups}
    path: list[str] = []
    for k in kind_chain(kind, onto):
        path.append(k.id)
        for entry in k.provides:
            if entry == action.value or action in groups.get(entry, []):
                return CapabilityReason(action=action, allowed=True, path=path, via=entry)
    return CapabilityReason(action=action, allowed=False, path=path)
```

```python
def fill_vehicle_kinds(sc: Scenario) -> Scenario:
```

```python
from contracts.models import ObjectGraph, ObjectLink, ObjectNode
from modules.gateway.engine_state import EngineState

TARGET_KO = {"boat": "보트", "fast_boat": "고속정", "person": "사람", "unknown": "미상"}
CHECK_KO = {"no_provenance": "출처", "outside_zone": "구역", "battery_insufficient": "배터리"}
LEAF = ("vehicle", "target")  # a vehicle links to all its commands; never expand through it


def _all(es: EngineState) -> tuple[dict[str, ObjectNode], list[ObjectLink]]:
    n: dict[str, ObjectNode] = {}
    ln: list[tuple[str | None, str | None, str]] = []
    for sp in es.specs.values():
        n[sp.id] = ObjectNode(id=sp.id, type="vehicle", label=f"{sp.id} · {(sp.kind_path or [sp.name])[0]}")
    for tg in es.targets:
        gone = " · 무력화" if tg.neutralized_ms else ""
        n[tg.id] = ObjectNode(id=tg.id, type="target", label=f"{tg.id} · {TARGET_KO[tg.kind.value]}{gone}",
                              t_ms=tg.neutralized_ms or tg.first_seen_ms)  # fmt: skip
    reps = {r.id: r for r in es.reports}
    for r in es.reports:
        n[r.id] = ObjectNode(id=r.id, type="report", label=r.text, t_ms=r.t_ms)
        ln += [(r.vehicle_id, r.id, "reported"), (r.id, r.target_id, "about")]
    for d in es.decisions:
        choice = next((c.label for c in reps[d.report_id].choices if c.id == d.choice_id), d.choice_id)
        n[d.id] = ObjectNode(id=d.id, type="decision", label=f"{d.id} '{choice}'", t_ms=d.t_ms)
        ln.append((d.id, d.report_id, "answers"))
    passed = {p.command.id: p for p in es.passed}
    for c in es.commands:
        p = passed.get(c.id)
        res = "·".join(CHECK_KO.get(x, x) for x in p.checks) + " 통과" if p else "차단"
        prov = c.provenance
        n[c.id] = ObjectNode(id=c.id, type="command", label=f"{c.kind.value} · {c.vehicle_id} · {res}",
                             t_ms=prov.issued_ms if prov else None)  # fmt: skip
        ln += [(c.id, c.vehicle_id, "sent_to"), (c.id, prov and prov.decision_id, "from_decision"),
               (c.id, c.params.get("target_id"), "aims_at")]  # fmt: skip
    for b in es.blocks:
        n[f"blk-{b.command_id}"] = ObjectNode(id=f"blk-{b.command_id}", type="block", label=b.detail, t_ms=b.t_ms)
        ln.append((f"blk-{b.command_id}", b.command_id, "blocked"))
    return n, [ObjectLink(src=s, dst=d, rel=r) for s, d, r in ln if s in n and d in n]


def build_object_graph(es: EngineState, focus: str) -> ObjectGraph:
    n, links = _all(es)
    if focus not in n:
        raise ValueError(f"없는 객체: {focus}")
    near: dict[str, list[str]] = {}
    for lk in links:
        near.setdefault(lk.src, []).append(lk.dst)
        near.setdefault(lk.dst, []).append(lk.src)
    seen, todo = {focus}, [focus]
    while todo:
        cur = todo.pop()
        if cur != focus and n[cur].type in LEAF:
            continue
        for nxt in near.get(cur, []):
            if nxt not in seen:
                seen.add(nxt)
                todo.append(nxt)
    return ObjectGraph(focus=focus, nodes=[x for x in n.values() if x.id in seen],
                       links=[lk for lk in links if lk.src in seen and lk.dst in seen])  # fmt: skip
```

```ts
import type { CapabilityReason, Ontology, OntologyView, Operation, Play, VehicleSpec } from "../../contracts.gen";

export interface KindRow { id: string; label: string; depth: number; adds: string; vehicles: string[] }

// The kind hierarchy as indented rows, each parent before its children, in the ontology's order (decision 0038).
export function kindRows(onto: Ontology, vehicles: VehicleSpec[]): KindRow[] {
  const groupLabel = new Map(onto.groups.map((g) => [g.id, g.label]));
  const rows: KindRow[] = [];
  const walk = (parent: string | null, depth: number) => {
    for (const k of onto.kinds.filter((x) => (x.is_a ?? null) === parent)) {
      rows.push({ id: k.id, label: k.label, depth, adds: (k.provides ?? []).map((p) => groupLabel.get(p) ?? p).join("·"),
        vehicles: vehicles.filter((v) => v.kind === k.id).map((v) => v.id) });
      walk(k.id, depth + 1);
    }
  };
  walk(null, 0);
  return rows;
}

// Why a vehicle can or cannot run one action: the kinds the server walked and what gave it.
export function reasonLine(r: CapabilityReason, onto: Ontology): string {
  const label = new Map(onto.kinds.map((k) => [k.id, k.label]));
  const path = r.path.map((id) => label.get(id) ?? id).join(" → ");
  if (!r.allowed) return `${path}, 어디에도 없음`;
  const group = onto.groups.find((g) => g.id === r.via);
  return `${path} (${group ? `${group.label} 묶음` : r.via})`;
}

// Actions of the plays a vehicle may be given: its start play, switch buttons and report choices naming it.
function actionsFor(vid: string, op: Operation, plays: Play[]): string[] {
  const ids = new Set<string>();
  if (op.assign?.[vid]) ids.add(op.assign[vid]);
  for (const sw of op.switches ?? []) if (sw.play_by_vehicle[vid]) ids.add(sw.play_by_vehicle[vid]);
  for (const p of plays) for (const s of p.steps) for (const c of s.report?.choices ?? []) {
    if (c.play_by_vehicle?.[vid]) ids.add(c.play_by_vehicle[vid]);
  }
  return [...new Set(plays.filter((p) => ids.has(p.id)).flatMap((p) => p.steps.map((s) => s.do.kind as string)))];
}

// One line per reserve; shown only, the engine's reserve choice is unchanged (decision 0038).
export function reserveLines(op: Operation, plays: Play[], view: OntologyView, vehicles: VehicleSpec[]): string[] {
  const kindOf = new Map(vehicles.map((v) => [v.id, v.kind ?? null]));
  const kindLabel = new Map(view.ontology.kinds.map((k) => [k.id, k.label]));
  const can = new Map(view.vehicles.map((v) => [v.vehicle_id, new Set<string>(v.reasons.filter((r) => r.allowed).map((r) => r.action))]));
  return Object.entries(op.reserve ?? {}).flatMap(([vid, reserves]) => {
    const need = actionsFor(vid, op, plays);
    return reserves.map((r) => {
      const missing = need.filter((a) => !can.get(r)?.has(a));
      if (missing.length) return `${r}는 ${vid}의 예비. ${vid}의 플레이가 쓰는 행동 중 ${missing.join("·")}를 ${r}가 못 함`;
      const kr = kindOf.get(r);
      const kind = kr && kr === kindOf.get(vid) ? `같은 ${kindLabel.get(kr)}` : `${r}: ${kindLabel.get(kr ?? "") ?? "종류 없음"}`;
      return `${r}는 ${vid}의 예비. ${vid}의 플레이가 쓰는 행동 ${need.length}개를 ${r}가 모두 할 수 있음(${kind})`;
    });
  });
}
```

```ts
import type { ObjectGraph, ObjectNode } from "../../contracts.gen";

export const COL_W = 260;
export const ROW_H = 84;
const COLUMN: Record<ObjectNode["type"], number> = { vehicle: 0, report: 1, decision: 2, command: 3, block: 3, target: 4 };
export const REL_KO: Record<string, string> = { reported: "보고함", about: "이 표적", answers: "답함",
  from_decision: "이 결정에서 나옴", sent_to: "이 기체에", aims_at: "겨눔", blocked: "막음" };

// Fixed columns by object type (vehicle, card, decision, command, target); in a column the earliest first,
// blocks after commands, objects without a time last (decision 0038).
export function layoutObjectGraph(g: ObjectGraph): Record<string, { x: number; y: number }> {
  const cols = new Map<number, ObjectNode[]>();
  for (const n of g.nodes) cols.set(COLUMN[n.type], [...(cols.get(COLUMN[n.type]) ?? []), n]);
  const out: Record<string, { x: number; y: number }> = {};
  for (const [c, nodes] of cols) {
    nodes.sort((a, b) => Number(a.type === "block") - Number(b.type === "block")
      || (a.t_ms ?? Number.MAX_SAFE_INTEGER) - (b.t_ms ?? Number.MAX_SAFE_INTEGER) || a.id.localeCompare(b.id));
    nodes.forEach((n, i) => { out[n.id] = { x: c * COL_W, y: i * ROW_H }; });
  }
  return out;
}
```

```tsx
import { Background, BackgroundVariant, type Edge, MarkerType, type Node, Position, ReactFlow } from "@xyflow/react";
import "@xyflow/react/dist/style.css";
import { useEffect, useState } from "react";
import type { ObjectGraph } from "../../contracts.gen";
import "./graph.css";
import { layoutObjectGraph, REL_KO } from "./objectGraphLayout";

const TYPE_KO = { vehicle: "기체", target: "표적", report: "보고 카드", decision: "결정", command: "명령", block: "차단" } as const;

// "연결 보기" (decision 0038): the objects linked to one target, fetched once from GET /graph when opened.
export function ObjectGraphView({ focus, onClose }: { focus: string; onClose: () => void }) {
  const [graph, setGraph] = useState<ObjectGraph | null>(null);
  const [error, setError] = useState<string | null>(null);
  useEffect(() => {
    fetch(`/graph?focus=${encodeURIComponent(focus)}`)
      .then(async (r) => {
        if (!r.ok) throw new Error(`/graph answered ${r.status}`);
        setGraph((await r.json()) as ObjectGraph);
      })
      .catch((e: Error) => setError(e.message));
  }, [focus]);
  const pos = graph ? layoutObjectGraph(graph) : {};
  const nodes: Node[] = (graph?.nodes ?? []).map((n) => ({
    id: n.id, position: pos[n.id], draggable: false, connectable: false,
    sourcePosition: Position.Right, targetPosition: Position.Left,
    className: `og-node og-${n.type}${n.id === focus ? " og-focus" : ""}`,
    data: { label: <><small>{TYPE_KO[n.type]}</small><div>{n.label}</div></> },
  }));
  const edges: Edge[] = (graph?.links ?? []).map((l, i) => ({
    id: `${l.src}-${l.rel}-${l.dst}-${i}`, source: l.src, target: l.dst, label: REL_KO[l.rel] ?? l.rel,
    markerEnd: { type: MarkerType.ArrowClosed, color: "#8e8e93" }, style: { stroke: "#8e8e93" },
    labelStyle: { fill: "#e5e5e5", fontSize: 11 }, labelBgStyle: { fill: "#000000", fillOpacity: 0.85 },
  }));
  return (
    <div className="object-graph">
      <div className="og-head">
        <b>{focus} 연결</b>
        <button onClick={onClose}>닫기</button>
      </div>
      {error ? <div className="og-note">연결을 불러오지 못했습니다 ({error})</div>
        : !graph ? <div className="og-note">불러오는 중…</div>
        : (
          <ReactFlow nodes={nodes} edges={edges} fitView fitViewOptions={{ padding: 0.15, maxZoom: 1 }}
            nodesDraggable={false} nodesConnectable={false} proOptions={{ hideAttribution: true }}>
            <Background variant={BackgroundVariant.Dots} gap={16} size={1} />
          </ReactFlow>
        )}
    </div>
  );
}
```
